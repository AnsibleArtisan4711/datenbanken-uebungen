=== Aufgabe 2: Filtern
SELECT vorname, nachname, abteilung_id FROM mitarbeiter WHERE abteilung_id = 2 AND nachname LIKE 'M%';
 vorname | nachname | abteilung_id 
---------+----------+--------------
 Peter   | Maier    |            2
 Anna    | Müller   |            2
(2 rows)

SELECT vorname, nachname, gehalt, abteilung_id FROM mitarbeiter WHERE abteilung_id = 2 AND nachname LIKE 'M%' OR gehalt > 4000;
 vorname | nachname | gehalt  | abteilung_id 
---------+----------+---------+--------------
 Peter   | Maier    | 4200.00 |            2
 Lukas   | Steiner  | 4500.00 |            3
 Anna    | Müller   | 3990.00 |            2
(3 rows)

SELECT vorname, nachname, gehalt, abteilung_id FROM mitarbeiter WHERE (abteilung_id = 2 AND nachname LIKE 'M%' OR gehalt > 4000) AND NOT abteilung_id = 3;
 vorname | nachname | gehalt  | abteilung_id 
---------+----------+---------+--------------
 Peter   | Maier    | 4200.00 |            2
 Anna    | Müller   | 3990.00 |            2
(2 rows)
