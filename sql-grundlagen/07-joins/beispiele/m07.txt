=== Ergänzung: Zeilenzahlen der Join-Arten
SELECT 'INNER' AS art, COUNT(*) FROM mitarbeiter m INNER JOIN abteilung a ON a.abteilung_id = m.abteilung_id
UNION ALL SELECT 'LEFT', COUNT(*) FROM mitarbeiter m LEFT JOIN abteilung a ON a.abteilung_id = m.abteilung_id
UNION ALL SELECT 'RIGHT', COUNT(*) FROM mitarbeiter m RIGHT JOIN abteilung a ON a.abteilung_id = m.abteilung_id
UNION ALL SELECT 'FULL', COUNT(*) FROM mitarbeiter m FULL JOIN abteilung a ON a.abteilung_id = m.abteilung_id;
  art  | count 
-------+-------
 INNER |     5
 LEFT  |     6
 RIGHT |     6
