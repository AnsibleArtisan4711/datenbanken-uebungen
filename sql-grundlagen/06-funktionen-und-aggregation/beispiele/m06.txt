=== Aufgabe 2: Aggregation

=== Aufgabe 3: GROUP BY und HAVING
SELECT abteilung_id, COUNT(*) AS anzahl FROM mitarbeiter GROUP BY abteilung_id ORDER BY abteilung_id;
 abteilung_id | anzahl 
--------------+--------
            1 |      1
            2 |      3
            3 |      1
(3 rows)

SELECT abteilung_id, COUNT(*) AS anzahl FROM mitarbeiter GROUP BY abteilung_id HAVING COUNT(*) > 1;
 abteilung_id | anzahl 
--------------+--------
            2 |      3
(1 row)
